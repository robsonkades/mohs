=== sem filtro (primeira pagina) ===
SELECT id, job_key, state, scheduled_at, fired_at, actor, priority, idempotency_key FROM mohs_executions ORDER BY id DESC LIMIT 51

SELECT
    "ID",
    "JOB_KEY",
    "STATE",
    "SCHEDULED_AT",
    "FIRED_AT",
    "ACTOR",
    "PRIORITY",
    "IDEMPOTENCY_KEY"
FROM "PUBLIC"."MOHS_EXECUTIONS"
    /* PUBLIC.PRIMARY_KEY_C */
ORDER BY 1 DESC
FETCH FIRST 51 ROWS ONLY

    "ID",
    "JOB_KEY",
    "STATE",
    "SCHEDULED_AT",
    "FIRED_AT",
    "ACTOR",
    "PRIORITY",

    /* PUBLIC.CONSTRAINT_INDEX_C: JOB_KEY = 'findpage-job-42' */
