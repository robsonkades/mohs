SELECT
    "E"."ID" AS "ID",
    "E"."JOB_KEY" AS "JOB_KEY",
    "J"."ALLOW_CONCURRENT_EXECUTIONS" AS "ALLOW_CONCURRENT_EXECUTIONS"
FROM "PUBLIC"."MOHS_EXECUTIONS" "E"
    /* PUBLIC.IDX_MOHS_EXECUTIONS_CLAIM: STATE = 'ENQUEUED'
        AND SCHEDULED_AT <= TIMESTAMP '2026-08-14 12:00:00'
     */
    /* WHERE (E.STATE = 'ENQUEUED')
        AND (E.SCHEDULED_AT <= TIMESTAMP '2026-08-14 12:00:00')
    */
INNER JOIN "PUBLIC"."MOHS_JOB_DEFINITIONS" "J"
    /* PUBLIC.CONSTRAINT_INDEX_7: JOB_KEY = E.JOB_KEY */
    ON 1=1
WHERE ("J"."JOB_KEY" = "E"."JOB_KEY")
    AND (("E"."STATE" = 'ENQUEUED')
    AND ("E"."SCHEDULED_AT" <= TIMESTAMP '2026-08-14 12:00:00'))
    AND (("J"."ALLOW_CONCURRENT_EXECUTIONS" = TRUE)
    OR ("J"."RUNNING_EXECUTION_COUNT" < "J"."MAX_CONCURRENT_EXECUTIONS"))
ORDER BY "E"."PRIORITY", "E"."SCHEDULED_AT"
