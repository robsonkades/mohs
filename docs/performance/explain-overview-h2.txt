=== backlog (ENQUEUED + RETRY_SCHEDULED, indice parcial de claim) ===
SELECT state, COUNT(*) AS total FROM mohs_executions WHERE state IN ('ENQUEUED', 'RETRY_SCHEDULED') GROUP BY state

SELECT
    "STATE",
    COUNT(*) AS "TOTAL"
FROM "PUBLIC"."MOHS_EXECUTIONS"
    /* PUBLIC.IDX_MOHS_EXECUTIONS_REAPER: STATE IN('ENQUEUED', 'RETRY_SCHEDULED') */
WHERE "STATE" IN('ENQUEUED', 'RETRY_SCHEDULED')
GROUP BY "STATE"

=== RUNNING (indice parcial do reaper via IS NOT NULL) ===
SELECT COUNT(*) FROM mohs_executions WHERE state = 'RUNNING' AND lease_expires_at IS NOT NULL

SELECT
    COUNT(*)
FROM "PUBLIC"."MOHS_EXECUTIONS"
    /* PUBLIC.IDX_MOHS_EXECUTIONS_REAPER: STATE = 'RUNNING' */
WHERE ("STATE" = 'RUNNING')
    AND ("LEASE_EXPIRES_AT" IS NOT NULL)

=== janela de vazao (idx_mohs_attempts_throughput) ===
SELECT outcome, COUNT(*) AS total FROM mohs_attempts WHERE finished_at >= TIMESTAMP '2026-08-17 11:59:00' AND outcome IN ('SUCCEEDED', 'FAILED') GROUP BY outcome

SELECT
    "OUTCOME",
    COUNT(*) AS "TOTAL"
FROM "PUBLIC"."MOHS_ATTEMPTS"
    /* PUBLIC.IDX_MOHS_ATTEMPTS_THROUGHPUT: FINISHED_AT >= TIMESTAMP '2026-08-17 11:59:00' */
WHERE ("OUTCOME" IN('SUCCEEDED', 'FAILED'))
    AND ("FINISHED_AT" >= TIMESTAMP '2026-08-17 11:59:00')
GROUP BY "OUTCOME"
